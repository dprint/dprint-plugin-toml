~~ preferSingleLine: true ~~
== should collapse an array and an inline table that fit ==
a = [
  1,
  2,
]
b = {
  c = 1,
  d = 2,
}

[expect]
a = [1, 2]
b = { c = 1, d = 2 }

== should keep an array that doesn't fit over several lines ==
a = ["aaaaaaaaaaaaaaaaaaaa", "bbbbbbbbbbbbbbbbbbbb", "cccccccccccccccccccc", "dddddddddddddddddddd", "eeeeeeeeeeeeeeeeeeee", "ffffffffffffffffffff"]

[expect]
a = [
  "aaaaaaaaaaaaaaaaaaaa",
  "bbbbbbbbbbbbbbbbbbbb",
  "cccccccccccccccccccc",
  "dddddddddddddddddddd",
  "eeeeeeeeeeeeeeeeeeee",
  "ffffffffffffffffffff",
]

== should keep an array holding a comment over several lines ==
a = [
  # about 1
  1,
  2,
]

[expect]
a = [
  # about 1
  1,
  2,
]
