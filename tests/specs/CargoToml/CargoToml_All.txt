-- Cargo.toml --
== should format "comit" Cargo.toml file ==
[package]
authors = ["Some author"]
name = "package"
description = "Description"
version = "0.1.0"
edition = "2018"

[dependencies]
b = "0.1"
a = { version = "0.8", features = ["serde", "v4"] }

[dev-dependencies]
c = "0.1"
a = { version = "0.8", features = ["serde", "v4"] }
b = "0.2"

[features]
default = []
test = ["feature"]

[expect]
[package]
name = "package"
version = "0.1.0"
authors = ["Some author"]
edition = "2018"
description = "Description"

[dependencies]
a = { version = "0.8", features = ["serde", "v4"] }
b = "0.1"

[dev-dependencies]
a = { version = "0.8", features = ["serde", "v4"] }
b = "0.2"
c = "0.1"

[features]
default = []
test = ["feature"]

== should allow having dependencies in groups based on a blank line ==
[dependencies]

b = "1"
m = "3"

u = "2"
a = "1"

[expect]
[dependencies]
b = "1"
m = "3"
z = "2"

a = "1"
u = "2"

== should not invent blank lines in an array when sorting moves an entry ==
[package]
zzz = "last"
keywords = [
  "macros",
  "syn",
]
name = "x"

[expect]
[package]
name = "x"
keywords = [
  "macros",
  "syn",
]
zzz = "last"
