~~ array.spaceSurroundingBrackets: true ~~
== should pad a single-line array ==
a = [1, 2]
b = []
c = { d = [1, 2], e = [] }

[expect]
a = [ 1, 2 ]
b = []
c = { d = [ 1, 2 ], e = [] }

== should not pad a multi-line array ==
a = [
  1,
  2,
]

[expect]
a = [
  1,
  2,
]

== should not double the space when a comment sits against a bracket ==
a = { b = [
# lead
1
] }
c = { d = [ # after open
1, 2] }

[expect]
a = { b = [
  # lead
  1,
] }
c = { d = [ # after open
  1,
  2,
] }
