== should only format the touched entry when the conventions don't move it ==
[package]
name   =   "a"
[|version   =   "0.1.0"|]
edition   =   "2021"

[dependencies]
b   =   "1"
a   =   "1"

[expect]
[package]
name   =   "a"
[|version = "0.1.0"|]
edition   =   "2021"

[dependencies]
b   =   "1"
a   =   "1"

== should format the table when the conventions move the touched entry ==
[package]
edition   =   "2021"
name   =   "a"

[dependencies]
[|b   =   "1"|]
a   =   "1"

[dev-dependencies]
d   =   "1"

[expect]

name   =   "a"

[dependencies]

[|b = "1"|]

[dev-dependencies]
d   =   "1"
c   =   "1"
