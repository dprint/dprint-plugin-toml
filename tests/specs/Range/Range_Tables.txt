[ a ]
[|[ b ]|]
[ c ]

[expect]
[ a ]
[|[b]|]
[ c ]

== should format a touched table header with a dotted key ==
[ a ]
[|[ b . "c d" . 'e' ]|]
[ c ]

[expect]
[ a ]
[|[b."c d".'e']|]
[ c ]

[[ c ]]

[expect]
[[ a ]]
[|[[b.c]]|]
[[ c ]]
