== should write the trailing comma before a trailing comment ==
a = [
  1, # one
  2 # two
]

[expect]
a = [
  1, # one
  2, # two
]

== should write the trailing comma before a comment on its own line ==
a = [
  1,
  2
  # done
]

[expect]
a = [
  1,
  2,
  # done
]

== should not write a trailing comma when the array stays on one line ==
a = [1, 2] # after

[expect]
a = [1, 2] # after

== should write the trailing comma in a multi-line inline table with comments ==
a = {
  b = 1, # one
  c = 2 # two
  # done
}

[expect]
a = {
  b = 1, # one
  c = 2, # two
  # done
}

== should write the trailing comma when only a comment forces the break ==
a = [1,
  # note
  2]

[expect]
a = [
  1,
  # note
  2,
]

== should write the trailing comma after a value carrying a comment above it ==
a = [
  1,
  # about two
  2
]

[expect]
a = [
  1,
  # about two
  2,
]
